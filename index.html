        const App = () => {

            const [projects, setProjects] = useState(initialProjects);

            const [view, setView] = useState('dashboard'); // dashboard, detail, email, approver

            const [activeProjectId, setActiveProjectId] = useState(null);

            

            // Modals state

            const [isNewProjectModalOpen, setIsNewProjectModalOpen] = useState(false);

            const [isNewVersionModalOpen, setIsNewVersionModalOpen] = useState(false);


            const activeProject = projects.find(p => p.id === activeProjectId);


            // --- Handlers ---

            const handleRequestProof = (data) => {

                const newId = `p${Date.now()}`;

                const newProject = {

                    id: newId,

                    name: data.name,

                    approverName: data.approverName || 'Unnamed Approver',

                    approverEmail: data.approverEmail,

                    dueDate: data.dueDate || 'No Date',

                    status: 'Pending Approval',

                    versions: [

                        {

                            id: `v${Date.now()}`,

                            versionNumber: 1,

                            sentDate: 'Just now',

                            message: data.message,

                            links: data.link ? [data.link] : [],

                            status: 'Pending Approval',

                            feedback: null

                        }

                    ]

                };

                setProjects([newProject, ...projects]);

                setIsNewProjectModalOpen(false);

            };


            const handleSendNewVersion = (data) => {

                const project = activeProject;

                const newVersionNum = project.versions.length + 1;

                

                const updatedProjects = projects.map(p => {

                    if (p.id !== project.id) return p;

                    

                    const newVersion = {

                        id: `v${Date.now()}`,

                        versionNumber: newVersionNum,

                        sentDate: 'Just now',

                        message: data.message,

                        links: data.link ? [data.link] : [],

                        status: 'Pending Approval',

                        feedback: null

                    };


                    return {

                        ...p,

                        status: 'Pending Approval', // Reset status

                        versions: [newVersion, ...p.versions] // Insert at beginning

                    };

                });

                

                setProjects(updatedProjects);

                setIsNewVersionModalOpen(false);

            };


            const handleApproverDecision = (projectId, decision, feedback = null) => {

                const updatedProjects = projects.map(p => {

                    if (p.id !== projectId) return p;

                    

                    // Update latest version status and feedback

                    const updatedVersions = [...p.versions];

                    updatedVersions[0] = {

                        ...updatedVersions[0],

                        status: decision,

                        feedback: feedback

                    };


                    return {

                        ...p,

                        status: decision, // Update overall project status

                        versions: updatedVersions

                    };

                });

                

                setProjects(updatedProjects);

            };


            const openProject = (project) => {

                setActiveProjectId(project.id);

                setView('detail');

            };


            // --- Render Logic ---

            return (

                <div className="min-h-screen">

                    {/* Navigation bar for Internal Views */}

                    {(view === 'dashboard' || view === 'detail') && (

                        <nav className="bg-slate-900 text-white sticky top-0 z-20">

                            <div className="max-w-6xl mx-auto px-4 h-14 flex items-center justify-between">

                                <div className="flex items-center space-x-2 font-bold text-lg cursor-pointer" onClick={() => setView('dashboard')}>

                                    <div className="w-6 h-6 bg-primary rounded flex items-center justify-center text-xs">IC</div>

                                    <span>Comms Proofs</span>

                                </div>

                                <div className="flex items-center gap-4 text-sm font-medium">

                                    <span className="text-white">Dashboard</span>

                                    <span className="text-slate-400 hover:text-white cursor-pointer transition-colors">Archive</span>

                                    <div className="w-8 h-8 rounded-full bg-slate-700 flex items-center justify-center border border-slate-600">JD</div>

                                </div>

                            </div>

                        </nav>

                    )}


                    {view === 'dashboard' && (

                        <Dashboard 

                            projects={projects} 

                            onProjectClick={openProject} 

                            onRequestProof={() => setIsNewProjectModalOpen(true)}

                        />

                    )}


                    {view === 'detail' && (

                        <ProjectDetail 

                            project={activeProject} 

                            onBack={() => setView('dashboard')}

                            onNewVersion={() => setIsNewVersionModalOpen(true)}

                            onSimulateEmail={() => setView('email')}

                        />

                    )}


                    {view === 'email' && (

                        <ApproverEmailPreview 

                            project={activeProject}

                            onSimulateClick={() => setView('approver')}

                            onBack={() => setView('detail')}

                        />

                    )}


                    {view === 'approver' && (

                        <ApproverPortal 

                            project={activeProject}

                            onDecision={handleApproverDecision}

                            onBackToComms={() => setView('detail')}

                        />

                    )}


                    {/* Modals rendered at app root */}

                    <RequestProofModal 

                        isOpen={isNewProjectModalOpen} 

                        onClose={() => setIsNewProjectModalOpen(false)}

                        onSubmit={handleRequestProof}

                    />

                    

                    <NewVersionModal 

                        isOpen={isNewVersionModalOpen}

                        onClose={() => setIsNewVersionModalOpen(false)}

                        onSubmit={handleSendNewVersion}

                        project={activeProject}

                    />

                </div>

            );

        };


        const root = ReactDOM.createRoot(document.getElementById('root'));

        root.render(<App />);
